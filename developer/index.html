📝 Quick Start

Install the SDK

// Install LNBQSHA SDK
npm install lnbqsha-sdk

// Import and use
import { LNBQSHA } from 'lnbqsha-sdk';

const client = new LNBQSHA({
  baseUrl: 'https://api.lnbqsha.com',
  apiKey: 'your-api-key'
});

// Get wallet balance
const wallet = await client.getWallet();
console.log(wallet);
